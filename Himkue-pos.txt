function pushCart(m,n,b){
  var ok=n+(b?'|'+b:''); var pr=gp(m);
  var ex=null;
  for(var i=0;i<cart.length;i++){ if(cart[i].mid==m.id&&cart[i].ok===ok){ex=cart[i];break;} }
  if(ex){ ex.qty++; }
  else{
    var p=[]; if(n)p.push(n); if(b)p.push(b);
    cart.push({k:Date.now()+Math.random(),mid:m.id,nm:m.name,em:m.emoji,pr:pr,qty:1,ok:ok,sb:p.join(' \u00b7 ')});
  }
  rCart();
}
function chg(k,d){

    if(cart[i].k===k){ cart[i].qty+=d; if(cart[i].qty<=0) cart.splice(i,1); break; }
  }
  rCart();
}

function dateRange(){
  var now=new Date();
  var from,to=now.toISOString().slice(0,10);
  if(period==='day'){
    from=to;
  } else if(period==='week'){
    var d=new Date(now.getTime()-6*24*60*60*1000);
    from=d.toISOString().slice(0,10);
  } else {
    from=now.getFullYear()+'-'+String(now.getMonth()+1).padStart(2,'0')+'-01';
  }
  return {from:from,to:to};
}

// ── DASHBOARD: OVERVIEW ───────────────────────
function rOverview(){
  var dr=dateRange();
  api({action:'getOrders',from:dr.from,to:dr.to},function(err,res){
    if(err||!res||!res.ok){ return; }
    orders=res.data||[];
    var tot=0,cnt=orders.length,cogs=0,wt=0,lt=0,gt=0;
    for(var i=0;i<orders.length;i++){
      var o=orders[i]; tot+=o.total;
      if(o.channel==='walkin') wt+=o.total;
      else if(o.channel==='lineman') lt+=o.total;
      else if(o.channel==='grab') gt+=o.total;
      for(var j=0;j<o.items.length;j++){
        var it=o.items[j]; var mn=null;
        for(var k=0;k<menu.length;k++){if(menu[k].name===it.name){mn=menu[k];break;}}
        if(mn&&mn.cost) cogs+=mn.cost*it.qty;
      }
    }
    var avg=cnt?Math.round(tot/cnt):0;
    document.getElementById('statGrid').innerHTML=
      mkSt('\ud83d\udcb0','\u0e22\u0e2d\u0e14\u0e23\u0e27\u0e21','\u0e3f'+tot.toLocaleString(),'g','')+
      mkSt('\ud83d\udccb','\u0e2d\u0e2d\u0e40\u0e14\u0e2d\u0e23\u0e4c',cnt+' \u0e2d\u0e2d\u0e40\u0e14\u0e2d\u0e23\u0e4c','gr','')+
      mkSt('\ud83e\udde0','\u0e40\u0e09\u0e25\u0e35\u0e48\u0e22/\u0e2d\u0e2d\u0e40\u0e14\u0e2d\u0e23\u0e4c','\u0e3f'+avg,'b','')+
      mkSt('\ud83d\udcca','Gross Profit','\u0e3f'+(tot-cogs).toLocaleString(),'r',(tot-cogs)>=0?'pos':'neg');
    var mx=Math.max(wt,lt,gt,1);
    document.getElementById('chBars').innerHTML=
      mkChB('walkin','\ud83c\udfe2 \u0e2b\u0e19\u0e49\u0e32\u0e23\u0e49\u0e32\u0e19',wt,mx)+
      mkChB('lineman','\ud83d\udfe2 LINE MAN',lt,mx)+
      mkChB('grab','\ud83d\udfe2 Grab',gt,mx);
    // hour chart
    var hrs=new Array(24).fill(0);
    for(var i=0;i<orders.length;i++){
      if(orders[i].ts){ var h=new Date(orders[i].ts).getHours(); hrs[h]+=orders[i].total; }
    }
    var mxH=Math.max.apply(null,hrs)||1; var hw='';
    for(var h=6;h<=22;h++){
      var p=Math.round(hrs[h]/mxH*100);
      var isPeak=(hrs[h]===Math.max.apply(null,hrs.slice(6,23)));
      hw+='<div class="hbw"><div class="hb'+(isPeak?' peak':'')+'" style="height:'+p+'%"></div><div class="hl">'+h+'</div></div>';
    }
    document.getElementById('hourWrap').innerHTML=hw;
    // top items
    var sm={};
    for(var i=0;i<orders.length;i++){
      for(var j=0;j<orders[i].items.length;j++){
        var it=orders[i].items[j];
        if(!sm[it.name]) sm[it.name]={qty:0,rev:0,em:''};
        sm[it.name].qty+=it.qty; sm[it.name].rev+=it.price*it.qty;
        for(var k=0;k<menu.length;k++){if(menu[k].name===it.name){sm[it.name].em=menu[k].emoji;break;}}
      }
    }
    var sr=Object.keys(sm).map(function(k){return{nm:k,qty:sm[k].qty,rev:sm[k].rev,em:sm[k].em};})
      .sort(function(a,b){return b.qty-a.qty;}).slice(0,7);
    var medals=['\ud83e\udd47','\ud83e\udd48','\ud83e\udd49'];
    var th='';
    for(var i=0;i<sr.length;i++){
      th+='<div class="titem">'+
          '<div class="tir">'+(medals[i]||i+1)+'</div>'+
          '<div class="tiem">'+(sr[i].em||'\ud83c\udf5c')+'</div>'+
          '<div class="tinm">'+sr[i].nm+'</div>'+
          '<div class="ticnt">x'+sr[i].qty+'</div>'+
          '<div class="tirev">\u0e3f'+sr[i].rev.toLocaleString()+'</div>'+
          '</div>';
    }
    document.getElementById('topItems').innerHTML=th||'<div style="color:var(--gray);font-size:12px;text-align:center;padding:8px">\u0e22\u0e31\u0e07\u0e44\u0e21\u0e48\u0e21\u0e35\u0e02\u0e49\u0e2d\u0e21\u0e39\u0e25</div>';
  });
}
function mkSt(ic,lb,v,col,cls){
  return '<div class="stc"><div class="stic '+col+'">'+ic+'</div><div><div class="stlb">'+lb+'</div><div class="stv '+(cls||'')+'">'+v+'</div></div></div>';

function mkChB(ch,lb,val,mx){
  var p=mx>0?Math.round(val/mx*100):0;
  return '<div class="chbr"><div class="chbr-l">'+lb+'</div>'+
         '<div class="chbr-t"><div class="chbr-f '+ch+'" style="width:'+p+'%">'+(p>14?'\u0e3f'+val.toLocaleString():'')+'</div></div>'+
         '<div class="chbr-a">\u0e3f'+val.toLocaleString()+'</div></div>';
}

         '<div class="scac">'+
         '<button class="scab a" onclick="openScF(\''+s.id+'\',\'a\')">\u25b2 \u0e23\u0e31\u0e1a</button>'+

         '<button class="scab e" onclick="editSt(\''+s.id+'\')">&#x270f;&#xfe0f;</button>'+
         '</div>'+
         '<div class="scfm" id="sf-'+s.id+'"><input class="scin" type="number" min="1" id="si-'+s.id+'" placeholder="\u0e08\u0e33\u0e19\u0e27\u0e19"><button class="scok" onclick="apScF(\''+s.id+'\')">OK</button></div>'+
         '</div>';
    }
    document.getElementById('stkGrid').innerHTML=h;
  });
}
function openScF(id,mode){
  var f=document.getElementById('sf-'+id);
  var same=scMode[id]===mode&&f.classList.contains('on');
  document.querySelectorAll('.scfm').forEach(function(e){e.classList.remove('on');});
  scMode={};
  if(!same){ f.classList.add('on'); scMode[id]=mode; document.getElementById('si-'+id).focus(); }
}
function apScF(id){
  var v=parseInt(document.getElementById('si-'+id).value);
  if(!v||v<=0) return;
  var delta=(scMode[id]==='a')?v:-v;
  api({action:'adjustStock',id:id,delta:delta},function(err,res){
    if(err||!res||!res.ok){toast('\u274c','\u0e1a\u0e31\u0e19\u0e17\u0e36\u0e01\u0e44\u0e21\u0e48\u0e44\u0e14\u0e49','r');return;}
    document.getElementById('si-'+id).value='';
    toast(delta>0?'\u2705':'\u2193','\u0e2d\u0e31\u0e1b\u0e40\u0e14\u0e15\u0e41\u0e25\u0e49\u0e27','g');
    rStockDash();
  });
}

function editSt(id){
  var s=null;
  for(var i=0;i<stock.length;i++){if(stock[i].id===id){s=stock[i];break;}}
  if(!s) return;
  stEd=id;
  document.getElementById('stT').textContent='\u270f\ufe0f \u0e41\u0e01\u0e49\u0e44\u0e02';
  document.getElementById('st-nm').value=s.name;
  document.getElementById('st-ct').value=s.cat;
  document.getElementById('st-u').value=s.unit;
  document.getElementById('st-q').value=s.qty;
  document.getElementById('st-mn').value=s.min;
  document.getElementById('st-mx').value=s.max;
  document.getElementById('st-em').value=s.emoji;
  document.getElementById('stMod').classList.add('on');
}
function closeStMod(){ document.getElementById('stMod').classList.remove('on'); stEd=null; }
function saveSt(){
  var nm=document.getElementById('st-nm').value.trim();
  var ct=document.getElementById('st-ct').value.trim();
  var u=document.getElementById('st-u').value.trim();
  var q=parseInt(document.getElementById('st-q').value)||0;
  var mn2=parseInt(document.getElementById('st-mn').value)||5;
  var mx=parseInt(document.getElementById('st-mx').value)||100;
  var em=document.getElementById('st-em').value.trim()||'\ud83d\udce6';
  if(!nm){toast('\u274c','\u0e01\u0e23\u0e38\u0e13\u0e32\u0e43\u0e2a\u0e48\u0e0a\u0e37\u0e48\u0e2d','r');return;}
  if(stEd){
    for(var i=0;i<stock.length;i++){
      if(stock[i].id===stEd){
        stock[i].name=nm;stock[i].cat=ct;stock[i].unit=u;
        stock[i].qty=q;stock[i].min=mn2;stock[i].max=mx;stock[i].emoji=em; break;
      }
    }
  } else {
    stock.push({id:'sx'+Date.now(),name:nm,cat:ct,emoji:em,qty:q,min:mn2,max:mx,unit:u});
  }
  api({action:'saveStock',items:stock},function(err,res){
    if(err||!res||!res.ok){toast('\u274c','\u0e1a\u0e31\u0e19\u0e17\u0e36\u0e01\u0e44\u0e21\u0e48\u0e44\u0e14\u0e49','r');return;}
    toast('\u2705','\u0e1a\u0e31\u0e19\u0e17\u0e36\u0e01\u0e40\u0e23\u0e35\u0e22\u0e1a\u0e23\u0e49\u0e2d\u0e22','g');
    closeStMod(); rStockDash();
  });
}

function toast(ic,msg,c){
  var el=document.getElementById('toasts');
  var t=document.createElement('div'); t.className='ti';
  var bg=c==='r'?'#C0392B':c==='g'?'#27AE60':'#C8922A';
  var tc=c==='gold'?'#2C1810':'#fff';
  t.style.cssText='background:'+bg+';color:'+tc;
  t.textContent=ic+' '+msg;
  el.appendChild(t);
  setTimeout(function(){
    t.style.opacity='0'; t.style.transition='opacity .3s';
    setTimeout(function(){if(t.parentNode)t.parentNode.removeChild(t);},300);
  },3000);
}
